104   /**
105    * 写入部门与人员期限：部门为空即清除，非空须存在（{@code DEPARTMENT_NOT_FOUND}）。正式员工不能设有效期；
106    * 外部、临时人员必须设，且在项目时区的今天至今天加各自上限天数之间（{@code permission.policy} 配置），
107    * 违反均抛 {@code ACCESS_WINDOW_INVALID}。截止日当天仍可用，到期时刻存为次日零点。
108    */
109   private void attributes(UserAccount user, Long departmentId, PersonType type, LocalDate day) {
110     user.setDepartment(
111         departmentId == null
112             ? null
113             : departments
114                 .findById(departmentId)
115                 .orElseThrow(() -> new ResourceNotFoundException("DEPARTMENT_NOT_FOUND", "部门不存在")));
116     if (type == PersonType.EMPLOYEE) {
117       if (day != null) throw new InvalidDataException("ACCESS_WINDOW_INVALID", "正式员工不设置临时有效期");
118       user.setExpiresAt(null);
119     } else {
120       if (day == null) throw new InvalidDataException("ACCESS_WINDOW_INVALID", "外部或临时人员必须设置有效期");
121       LocalDate today = LocalDate.now(policy.zoneId());
122       int maximum =
123           type == PersonType.EXTERNAL ? policy.externalMaxDays() : policy.temporaryMaxDays();
124       if (day.isBefore(today) || day.isAfter(today.plusDays(maximum)))
125         throw new InvalidDataException("ACCESS_WINDOW_INVALID", "有效期须在今天至 " + maximum + " 天以内");
126       user.setExpiresAt(day.plusDays(1).atStartOfDay(policy.zoneId()).toInstant());
127     }
128     user.setPersonType(type);
129   }

 68    * 沿上级链向上不能回到自身，以自身为上级也算成环（{@code PERMISSION_CYCLE}）；已有下级的菜单不能改为按钮
 69    * （冲突 {@code PERMISSION_HAS_CHILDREN}）；按钮不能填菜单地址（{@code BUTTON_PATH_NOT_ALLOWED}）。
 70    * 全部通过后才写入名称、类型、地址和上级，空白地址存为 null。
 71    */
 72   private void apply(Permission p, String name, PermissionType type, String path, Long parentId) {
 73     Permission parent = parentId == null ? null : require(parentId);
 74     if (type == PermissionType.BUTTON && parent == null)
 75       throw new InvalidDataException("BUTTON_PARENT_REQUIRED", "按钮必须隶属菜单");
 76     if (parent != null
 77         && (parent.getType() != PermissionType.MENU
 78             || !parent.getSystem().getId().equals(p.getSystem().getId())))
 79       throw new InvalidDataException("PERMISSION_PARENT_INVALID", "父节点必须是同一应用的菜单");
 80     for (Permission next = parent; next != null; next = next.getParent())
 81       if (Objects.equals(next.getId(), p.getId())) throw new InvalidDataException("PERMISSION_CYCLE", "权限树不能形成循环");
 82     if (type == PermissionType.BUTTON
 83         && p.getId() != null
 84         && permissions.existsByParent_Id(p.getId())) throw new ConflictException("PERMISSION_HAS_CHILDREN", "有子节点的菜单不能改为按钮");
 85     if (type == PermissionType.BUTTON && path != null && !path.isBlank())
 86       throw new InvalidDataException("BUTTON_PATH_NOT_ALLOWED", "按钮不设置菜单地址");
 87     p.setName(name);
 88     p.setType(type);
 89     p.setMenuPath(path == null || path.isBlank() ? null : path);
 90     p.setParent(parent);
 91   }

213             .toList());

  9 public final class RoleGraph {

 11 
 12   /** 防御性复制快照，领域对象不修改持久化实体。 */
 13   public RoleGraph(Map<Long, List<Node>> edges) {
 14     Map<Long, List<Node>> copy = new LinkedHashMap<>();
 15     edges.forEach((id, nodes) -> copy.put(id, List.copyOf(nodes)));
 16     this.edges = Collections.unmodifiableMap(copy);
 17   }

 19   /** 新增边的无环不变量由图对象自身维护。 */
 20   public void assertCanInherit(Long senior, Long inherited) {
 21     if (senior.equals(inherited)) throw new InvalidDataException("ROLE_SELF_INHERITANCE", "角色不能继承自身");
 22     Set<Long> seen = new HashSet<>();
 23     Deque<Long> pending = new ArrayDeque<>();
 24     pending.add(inherited);
 25     while (!pending.isEmpty()) {
 26       Long id = pending.removeFirst();
 27       if (id.equals(senior)) throw new InvalidDataException("ROLE_INHERITANCE_CYCLE", "该关系会形成循环继承");
 28       if (seen.add(id)) edges.getOrDefault(id, List.of()).forEach(n -> pending.addLast(n.id()));
 29     }
 30   }
 31 
 32   /** 图以有效路径对象承载遍历状态，返回最短且不重复的到达结果。 */
 33   public List<Path> effectivePaths(Node root) {
 34     List<Path> found = new ArrayList<>();
 35     Set<Long> seen = new HashSet<>();
 36     Deque<Traversal> pending = new ArrayDeque<>();
 37     pending.add(new Traversal(root, List.of(root.id())));
 38     while (!pending.isEmpty()) {
 39       Traversal current = pending.removeFirst();
 40       if (!current.node.enabled() || !seen.add(current.node.id())) continue;
 41       found.add(new Path(current.node.id(), current.ids));
 42       for (Node child : edges.getOrDefault(current.node.id(), List.of()))
 43         pending.addLast(current.extend(child));
 44     }
 45     return List.copyOf(found);
 46   }
 47 
 48   /** 一条有效遍历分支，不与外部分支共享可变路径。 */
 49   private record Traversal(Node node, List<Long> ids) {
 50     /** 扩展到子节点，不修改当前路径。 */
 51     Traversal extend(Node next) {
 52       List<Long> copy = new ArrayList<>(ids);
 53       copy.add(next.id());
 54       return new Traversal(next, List.copyOf(copy));
 55     }
 56   }

139             "SYNC_ADMINISTRATOR_CONFLICT", "用户名 " + row.username() + " 是管理员账号，员工信息不能覆盖它");
140       else if (differs(user, row, department)) {
141         if (write) apply(user, row, department);
142         usersUpdated++;
143       } else usersUnchanged++;
144     }
145     if (write) users.saveAll(created);
146 
147     // 步骤 4：按分公司补授角色，只增不减。
148     Map<Long, Set<Long>> holders = new HashMap<>();
149     for (Role role : roleByCode.values())
150       holders.put(role.getId(), new HashSet<>(userRoles.holders(role.getId())));
151     List<UserRole> grants = new ArrayList<>();
152     for (DirectoryData.Row row : rows) {
153       String code = rule.roleFor(row.company());
154       if (code == null) continue;
155       Role role = roleByCode.get(code);
156       UserAccount user = people.get(row.username());
157       if (user.getId() == null || !holders.get(role.getId()).contains(user.getId()))
158         grants.add(new UserRole(user, role));
159     }
160     if (write) userRoles.saveAll(grants);
161 
162     long companies = rows.stream().map(DirectoryData.Row::company).distinct().count();
163     if (write)
164       log.info(
165           "员工信息同步完成：{} 行，新建部门 {}、人员 {}，更新人员 {}，补授角色 {}",
166           rows.size(),
167           departmentsCreated,
168           created.size(),
169           usersUpdated,
170           grants.size());
171     return new DirectoryData.Summary(
172         rows.size(),
173         (int) companies,
174         departmentsCreated,
175         departmentsKept,
176         created.size(),
177         usersUpdated,
178         usersUnchanged,
179         grants.size(),
180         digest,
181         write);
182   }
183 
184   /** 新部门：写入时立即保存并取默认编码；预览时只构造对象。 */
185   private Department create(String name, Department parent, boolean write) {
186     Department value = new Department();
187     value.setName(name);
188     value.setParent(parent);
189     if (!write) return value;
190     departments.save(value);
191     value.setCode(DepartmentData.defaultCode(value.getId()));
192     return value;
193   }
194 
195   /** 来源字段（姓名、部门、分公司、来源部门、岗位、电话）是否与员工表不同，决定计入“更新”还是“未变化”。 */
196   private static boolean differs(UserAccount user, DirectoryData.Row row, Department department) {
197     return !Objects.equals(user.getDisplayName(), row.name())
198         || !Objects.equals(user.getDepartmentId(), department.getId())
199         || !Objects.equals(user.getCompanyName(), row.company())
200         || !Objects.equals(user.getSourceDepartment(), row.department())
201         || !Objects.equals(user.getJobTitle(), row.job())
202         || !Objects.equals(user.getPhone(), row.phone());
203   }
204 
205   /** 只覆盖员工表的来源字段；密码、启用状态、人员类型、有效期和角色授权保持不变。 */
206   private static void apply(UserAccount user, DirectoryData.Row row, Department department) {
207     user.setDisplayName(row.name());
208     user.setDepartment(department);
209     user.setCompanyName(row.company());
210     user.setSourceDepartment(row.department());
211     user.setJobTitle(row.job());
212     user.setPhone(row.phone());
213   }

 30   public void placeDepartments(
 31       Collection<DepartmentNode> existing, UnaryOperator<DepartmentNode> create) {
 32     Map<String, DepartmentNode> known = new HashMap<>();
 33     for (DepartmentNode node : existing) known.put(key(node.getParentId(), node.getName()), node);
 34     for (DirectoryData.Row row : rows) {
 35       Long company = place(known, null, row.company(), row.company(), create);
 36       place(known, company, row.department(), departmentKey(row), create);
 37     }

 40   /**
 41    * 放置一个分公司或部门并返回其标识：同一行键只放置并计数一次；按“上级 + 名称”找到已有部门即复用，否则交给 create 保存后计入新建。
 42    */
 43   private Long place(
 44       Map<String, DepartmentNode> known,
 45       Long parent,
 46       String name,
 47       String rowKey,
 48       UnaryOperator<DepartmentNode> create) {
 49     Long placed = departmentIds.get(rowKey);
 50     if (placed != null) return placed;
 51     DepartmentNode node = known.get(key(parent, name));
 52     if (node == null) {
 53       node = new DepartmentNode();
 54       node.setName(name);
 55       node.setParentId(parent);
 56       node = create.apply(node);
 57       known.put(key(parent, name), node);
 58       departmentsCreated++;
 59     } else departmentsKept++;
 60     departmentIds.put(rowKey, node.getId());
 61     return node.getId();

 70     List<Account> changed = new ArrayList<>();
 71     for (DirectoryData.Row row : rows) {
 72       Account account = existing.get(row.username());
 73       boolean fresh = account == null;
 74       if (fresh) {
 75         account = Account.unprovisioned(row.username());
 76         existing.put(row.username(), account);
 77       }

 82               row.company(),
 83               row.department(),
 84               row.job(),
 85               row.phone());

 87       else if (modified) usersUpdated++;
 88       else usersUnchanged++;

110         result.add(new Grant(row.username(), role));
111     }
112     roleGrants = result.size();
113     return result;
114   }
