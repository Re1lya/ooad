 74   @Transactional
 75   public UserResult create(CreateUserCommand input) {
 76     if (users.existsByUsername(input.getUsername())) throw new ConflictException("登录名已存在");
 77     PasswordPolicy.validate(input.getPassword());
 78     UserAccount user = CloneFactory.copy(new UserAccount(), input);
 79     user.setPasswordHash(encoder.encode(input.getPassword()));
 80     attributes(
 81         user, input.getDepartmentId(), PersonType.valueOf(input.getKind()), input.getExpiresOn());
 82     return UserViews.view(users.save(user));
 83   }

 85   @Transactional
 86   public UserResult update(Long id, UserData.Update input) {
 87     UserAccount user = lock(id);
 88     if (user.getAdministrator()
 89         && (!input.personType().equals(PersonType.EMPLOYEE.name()) || input.expiresOn() != null))
 90       throw new ForbiddenException("管理员不能改为有期限的人员身份");
 91     user.setDisplayName(input.displayName());
 92     attributes(
 93         user, input.departmentId(), PersonType.valueOf(input.personType()), input.expiresOn());
 94     return UserViews.view(user);
 95   }

 97   private void attributes(UserAccount user, Long departmentId, PersonType type, LocalDate day) {
 98     user.setDepartment(
 99         departmentId == null
100             ? null
101             : departments
102                 .findById(departmentId)
103                 .orElseThrow(() -> new ResourceNotFoundException("部门不存在")));
104     if (type == PersonType.EMPLOYEE) {
105       if (day != null) throw new IllegalArgumentException("正式员工不设置临时有效期");
106       user.setExpiresAt(null);
107     } else {
108       if (day == null) throw new IllegalArgumentException("外部或临时人员必须设置有效期");
109       LocalDate today = LocalDate.now(policy.zoneId());
110       int maximum =
111           type == PersonType.EXTERNAL ? policy.externalMaxDays() : policy.temporaryMaxDays();
112       if (day.isBefore(today) || day.isAfter(today.plusDays(maximum)))
113         throw new IllegalArgumentException("有效期须在今天至 " + maximum + " 天以内");
114       user.setExpiresAt(day.plusDays(1).atStartOfDay(policy.zoneId()).toInstant());
115     }
116     user.setPersonType(type);
117   }

  9   private final Map<Long, List<Node>> edges;

 11   /** 防御性复制快照，领域对象不修改持久化实体。 */
 12   public RoleGraph(Map<Long, List<Node>> edges) {
 13     Map<Long, List<Node>> copy = new LinkedHashMap<>();
 14     edges.forEach((id, nodes) -> copy.put(id, List.copyOf(nodes)));
 15     this.edges = Collections.unmodifiableMap(copy);
 16   }

 18   /** 新增边的无环不变量由图对象自身维护。 */
 19   public void assertCanInherit(Long senior, Long inherited) {
 20     if (senior.equals(inherited)) throw new IllegalArgumentException("角色不能继承自身");
 21     Set<Long> seen = new HashSet<>();
 22     Deque<Long> pending = new ArrayDeque<>();
 23     pending.add(inherited);
 24     while (!pending.isEmpty()) {
 25       Long id = pending.removeFirst();
 26       if (id.equals(senior)) throw new IllegalArgumentException("该关系会形成循环继承");
 27       if (seen.add(id)) edges.getOrDefault(id, List.of()).forEach(n -> pending.addLast(n.id()));
 28     }
 29   }
 30 
 31   /** 图以有效路径对象承载遍历状态，返回最短且不重复的到达结果。 */
 32   public List<Path> effectivePaths(Node root) {
 33     List<Path> found = new ArrayList<>();
 34     Set<Long> seen = new HashSet<>();
 35     Deque<Traversal> pending = new ArrayDeque<>();
 36     pending.add(new Traversal(root, List.of(root.id())));
 37     while (!pending.isEmpty()) {
 38       Traversal current = pending.removeFirst();
 39       if (!current.node.enabled() || !seen.add(current.node.id())) continue;
 40       found.add(new Path(current.node.id(), current.ids));
 41       for (Node child : edges.getOrDefault(current.node.id(), List.of()))
 42         pending.addLast(current.extend(child));
 43     }
 44     return List.copyOf(found);
 45   }
 46 
 47   /** 一条有效遍历分支，不与外部分支共享可变路径。 */
 48   private record Traversal(Node node, List<Long> ids) {
 49     /** 扩展到子节点，不修改当前路径。 */
 50     Traversal extend(Node next) {
 51       List<Long> copy = new ArrayList<>(ids);
 52       copy.add(next.id());
 53       return new Traversal(next, List.copyOf(copy));
 54     }
